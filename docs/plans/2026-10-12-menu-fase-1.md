# Menu START — fase 1 (base das telas) — plano de implementação

**Objetivo:** menu START no estilo do FireRed (com barra de ajuda) sobre a sala, abrindo OPÇÕES e CONTATO, com abertura automática na 1ª visita, dica "START: MENU" e links diretos (`#/contato`).

**Arquitetura:** uma pilha de telas dentro do cartucho (`carts/center/ui.js`): START empilha o menu, A empilha a tela escolhida, B desempilha; a sala fica congelada com a pilha aberta. Cada janela segue os números do decomp (`pokefirered`): posição, moldura, 15 px por item, cursor ▶. O console (`gba.js`) ganha `api.link`, `api.setLang`, `api.lang`, `api.sound` e chama `cart.open(path)` com o endereço.

**Tecnologia:** HTML/CSS/JS (módulos ES), sem dependências nem build de JS. Assets por `tools/build_assets.py` (Python stdlib). Testes: `node check.mjs` (lógica pura e dados) e `node tools/e2e.mjs` (Edge headless via CDP).

**Spec:** `docs/specs/2026-10-12-telas-e-menu-design.md` (fase 1 da seção 10). Contexto do console e da sala: `docs/specs/2026-10-12-centro-pokemon-design.md`.

```js
// Screens with their own link: 'projetos/3' (or '#/projetos/3') -> { screen: 'projetos', n: 3 }.
// Anything else -> null, which means the room.
export const LINKED = ['projetos', 'sobre', 'experiencia', 'habilidades', 'contato'];
export function route(path, projects) {
  const [screen, n, ...rest] = String(path ?? '').replace(/^#?\/?/, '').split('/');
  if (!LINKED.includes(screen) || rest.length) return null;
  if (n === undefined) return { screen, n: null };
  const i = Number(n);
  return screen === 'projetos' && Number.isInteger(i) && i >= 1 && i <= projects ? { screen, n: i } : null;
}

// Width in game pixels of one line in the game font (assets/font/widths.json from the build).
export const textWidth = (s, widths) => [...s].reduce((w, ch) => w + (widths[ch] ?? 6), 0);

// Text that has to fit a window: at most `lines` lines, each at most `max` px.
export function fitsLines(text, widths, max, lines) {
  const all = text.split('\n');
  return all.length <= lines && all.every((l) => textWidth(l, widths) <= max);
}

// FireRed prints one letter, then waits 8 / 4 / 1 frames (sTextSpeedFrameDelays): letters per second.
export const TEXT_SPEEDS = { slow: 59.7275 / 9, mid: 59.7275 / 5, fast: 59.7275 / 2 };

// Script multichoice window (script_menu.c): width from the longest item, height by item count.
const MULTI_HEIGHT = [0, 2, 4, 6, 7, 9, 11, 13, 14];
export const multiTiles = (maxWidth, count) => ({ w: Math.floor((maxWidth + 9) / 8) + 1, h: MULTI_HEIGHT[count] });
```

```js
// Windows, lists and the stack of screens the START menu opens, drawn the way FireRed's menus are.
import { esc } from '../../logic.js';

// A standard window: interior w×h at (x, y) in game px, the selected frame 8px around it
// (WindowFunc_DrawStandardFrame). Children are placed from the interior's top-left.
export function win(x, y, w, h, cls = '') {
  const el = document.createElement('div');
  el.className = `win ${cls}`.trim();
  Object.assign(el.style, { left: `${x - 8}px`, top: `${y - 8}px`, width: `${w + 16}px`, height: `${h + 16}px` });
  return el;
}

// Text placed like the game's printer: the glyph cell top on (x, y), lines 15px apart.
export function text(str, x, y, cls = '') {
  const p = document.createElement('p');
  p.className = `txt ${cls}`.trim();
  Object.assign(p.style, { left: `${x}px`, top: `${y}px` });
  p.innerHTML = str.split('\n').map((l) => `<span class="line">${esc(l)}</span>`).join('');
  return p;
}

// A list with FireRed's ▶ cursor (x 0, items at x 8, `pitch` px apart). Keys move it with SE_SELECT;
// the mouse over an item moves it silently; a tap or click on another item moves it, and A or a
// click on the pointed item chooses it. Items are buttons and the pointed one has focus, so screen
// readers follow the cursor.
export function createList(box, { labels, y = 0, pitch, wrap, start = 0, api, onMove, onChoose }) {
  let cur = -1;
  const cursor = text('▶', 0, y, 'cursor');
  const choose = () => {
    api.sfx('select');
    onChoose(cur);
  };
  const buttons = labels.map((label, i) => {
    const b = document.createElement('button');
    b.type = 'button';
    b.tabIndex = -1;
    b.className = 'item';
    b.style.top = `${y + pitch * i}px`;
    b.innerHTML = `<span class="line">${esc(label)}</span>`;
    b.addEventListener('pointerover', (e) => { if (e.pointerType === 'mouse') set(i, false); });
    b.addEventListener('click', (e) => {
      if (e.detail === 0) return;   // keyboard activation comes through button() instead
      if (i === cur) choose();
      else set(i, true);
    });
    return b;
  });
  box.append(cursor, ...buttons);

  function set(i, sound) {
    if (i === cur) return;
    if (sound) api.sfx('select');
    buttons[cur]?.removeAttribute('aria-current');
    cur = i;
    buttons[i].setAttribute('aria-current', 'true');
    buttons[i].focus({ preventScroll: true });
    cursor.style.top = `${y + pitch * i}px`;
    onMove?.(i);
  }
  set(Math.max(0, Math.min(start, labels.length - 1)), false);

  return {
    get index() { return cur; },
    focus: () => buttons[cur].focus({ preventScroll: true }),
    button(name) {
      const d = name === 'down' ? 1 : name === 'up' ? -1 : 0;
      if (d) {
        const n = labels.length;
        const next = wrap ? (cur + d + n) % n : Math.min(n - 1, Math.max(0, cur + d));
        if (next !== cur) set(next, true);
        return true;
      }
      if (name === 'a') {
        choose();
        return true;
      }
      return false;
    },
  };
}

// The screens over the room. The top one gets the buttons (each screen decides what B does).
// A click on the room around the windows goes to `outside`; `path` is the link of the topmost
// screen that has one ('' when none).
export function createStack(root, { onChange, outside }) {
  const layer = document.createElement('div');
  layer.className = 'ui';
  root.append(layer);
  const list = [];
  const top = () => list[list.length - 1] ?? null;
  layer.addEventListener('pointerdown', (e) => {
    if (e.target !== layer || !list.length) return;
    e.stopPropagation();   // not a click on the room
    outside();
  });
  return {
    get size() { return list.length; },
    get top() { return top(); },
    push(screen) {
      list.push(screen);
      layer.append(screen.el);
      screen.focus?.();
      onChange();
    },
    pop() {
      list.pop()?.el.remove();
      top()?.refresh?.();
      top()?.focus?.();
      onChange();
    },
    clear() {
      while (list.length) list.pop().el.remove();
      onChange();
    },
    button(name, down) {
      top()?.button(name, down);
    },
    path: () => [...list].reverse().find((s) => s.path)?.path ?? '',
  };
}
```

```js
// CONTATO, the way FireRed asks with a list (script_menu.c): the question stays in the message box
// and the choices sit in a window at the top left. E-MAIL asks again in a small window where the
// game puts YES/NO (interior ending at x=216, from y=72).
import { multiTiles, textWidth } from '../../../logic.js';
import { win, text, createList } from '../ui.js';

export function createContact({ api, widths, contact, say, close }) {
  const el = document.createElement('div');
  el.className = 'contact';
  const ask = document.createElement('div');
  ask.className = 'msg';
  ask.append(text(api.talk('contact_ask')[0], 16, 9));
  const labels = [...contact.map((c) => c.label), api.t('menu_sair')];
  const size = multiTiles(Math.max(...labels.map((l) => textWidth(l, widths))), labels.length);
  const box = win(8, 8, size.w * 8, size.h * 8);
  el.append(ask, box);
  const list = createList(box, { labels, y: 2, pitch: 14, wrap: labels.length > 3, api, onChoose: pick });
  let sub = null;

  function pick(i) {
    const c = contact[i];
    if (!c) return close();
    if (c.url.startsWith('mailto:')) return askMail(c.url);
    window.open(c.url, '_blank', 'noopener');
  }

  function askMail(url) {
    const options = [api.t('contact_write'), api.t('contact_copy')];
    const s = multiTiles(Math.max(...options.map((l) => textWidth(l, widths))), 2);
    const small = win(216 - s.w * 8, 72, s.w * 8, s.h * 8);
    el.append(small);
    sub = {
      box: small,
      list: createList(small, {
        labels: options, y: 2, pitch: 14, wrap: false, api,
        onChoose(j) {
          closeMail();
          if (j === 0) location.href = url;
          else copy(url.slice('mailto:'.length));
        },
      }),
    };
  }

  function closeMail() {
    sub.box.remove();
    sub = null;
    list.focus();
  }

  async function copy(address) {
    try {
      await navigator.clipboard.writeText(address);
      say(api.talk('copied'));
    } catch {
      say([address]);   // no clipboard: show it so it can be copied by hand
    }
  }

  return {
    el,
    path: 'contato',
    focus: () => (sub ? sub.list : list).focus(),
    button(name, down) {
      if (!down) return;
      if (name === 'b') {
        api.sfx('select');
        return sub ? closeMail() : close();
      }
      (sub ? sub.list : list).button(name);
    },
  };
}
```
